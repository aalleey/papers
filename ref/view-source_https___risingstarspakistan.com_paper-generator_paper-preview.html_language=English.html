function shuffleArray(arr) {
const copy = JSON.parse(JSON.stringify(arr || []));
for (let i = copy.length - 1; i > 0; i--) {
const j = Math.floor(Math.random() * (i + 1));
[copy[i], copy[j]] = [copy[j], copy[i]];
}
return copy;
}

// ✅ Shuffles MCQs strictly WITHIN each portion (a with a, b with b, etc.)
function shuffleMcqsWithPortions(mcqs) {
if (!mcqs || !mcqs.length) return [];
const hasPortions = mcqs.some(q => q.portion);
if (!hasPortions) return shuffleArray(mcqs);

const portionMap = {};
const unassigned = [];

mcqs.forEach(q => {
if (q.portion) {
const key = q.portion.toString().toLowerCase().trim();
if (!portionMap[key]) portionMap[key] = [];
portionMap[key].push(q);
} else {
unassigned.push(q);
}
});

let result = [];
// Maintain ascending order of portions (a, b, c...) but shuffle items inside each portion
Object.keys(portionMap).sort().forEach(key => {
result = result.concat(shuffleArray(portionMap[key]));
});

return result.concat(shuffleArray(unassigned));
}

// Shuffles questions strictly WITHIN their respective question groups (Q2, Q3...)
function shuffleGroupedQuestions(questions) {
if (!questions || !questions.length) return [];
const groups = {};
questions.forEach(q => {
const key = q.questionNumber || 'default';
if (!groups[key]) groups[key] = [];
groups[key].push(q);
});
let result = [];
Object.keys(groups).forEach(key => {
result = result.concat(shuffleArray(groups[key]));
});
return result;
}

// 2. Otherwise → auto-detect based on content

async function compressImage(file, maxSizeKB = 100) {
return new Promise((resolve, reject) => {
const reader = new FileReader();
reader.readAsDataURL(file);
reader.onload = event => {
const img = new Image();
img.src = event.target.result;
img.onload = () => {
const canvas = document.createElement('canvas');
const ctx = canvas.getContext('2d');
const MAX_WIDTH = 400;
let width = img.width; let height = img.height;
if (width > MAX_WIDTH) { height *= MAX_WIDTH / width; width = MAX_WIDTH; }
canvas.width = width; canvas.height = height;
ctx.drawImage(img, 0, 0, width, height);
resolve(canvas.toDataURL('image/jpeg', 0.9));
};
};
});
}

/**
* Returns HTML for MCQs with page break if needed
*/
function getMcqSection() {
if (!paperData.mcqQuestions || paperData.mcqQuestions.length === 0) return '';

const mode = paperSettings.languageMode;
const isUrduMode = mode === "Urdu";
const isBilingual = mode === "Bilingual";
const isFederal = paperSettings.headerStyleId === 'style2';

const { eng: sEng, urdu: sUrdu } = getDisplayText(paperData.mcqSectionHeading, paperData.mcqSectionHeadingUrdu, true);
// Suffix calculation (Marks / Time)
let mcqSuffix = '';
if (typeof paperData.mcqSectionHeadingSuffix === 'string') {
mcqSuffix = paperData.mcqSectionHeadingSuffix;
} else if (paperSettings.useBoardHeadings && paperData.boardPattern) {
const pattern = paperData.boardPattern;
const mcqTime = pattern?.timeObjective ? ` (Time: ${pattern.timeObjective})` : '';
let mcqMarksLabel = '';
if (pattern?.questionHeadings?.mcqs) {
const match = pattern.questionHeadings.mcqs.match(/\((\d+\s*Marks)\)/i);
if (match) mcqMarksLabel = ` (${match[1]})`;
}
mcqSuffix = mcqTime + mcqMarksLabel;
}

const title = combineToCenteredHeadingHtml(sEng, sUrdu, 'mcqSectionHeading') +
`<span class="mcq-heading-suffix" contenteditable="true" data-field="mcqSectionHeadingSuffix"> ${mcqSuffix}</span>`;
const { eng: subEng, urdu: subUrdu } = getDisplayText(paperData.mcqSubHeading, paperData.mcqSubHeadingUrdu, true);
let subHeadingHtml = '';
if (mode === "Bilingual") {
if (subEng || subUrdu) subHeadingHtml = `<div class='bilingual-sub-heading' contenteditable="true" data-field="mcqSubHeading"><span>${subEng}</span><span class='urdu-text' style="font-size: 0.9em;">${subUrdu}</span></div>`;
} else {
const hText = isUrduMode ? subUrdu || subEng : subEng || subUrdu;
const urduClass = isUrduMode ? 'urdu-text' : '';
if (hText) subHeadingHtml = `<div class='question-heading ${urduClass}' style='justify-content: center; font-style: italic; text-align: ${isUrduMode ? 'right' : 'left'};' contenteditable="true" data-field="mcqSubHeading">${hText}</div>`;
}

// Check if any MCQ contains a portion
const hasPortions = paperData.mcqQuestions.some(q => q.portion && q.portion.toString().trim() !== "");

// Grouping by portion
const getGroupedMcqs = () => {
if (!hasPortions) {
return [{ portionKey: '', heading: '', headingUrdu: '', items: paperData.mcqQuestions }];
}
const map = {};
const unassigned = [];
paperData.mcqQuestions.forEach(q => {
if (q.portion && q.portion.toString().trim() !== "") {
const key = q.portion.toString().toLowerCase().trim();
if (!map[key]) {
map[key] = {
portionKey: key,
heading: q.portionHeading || `Choose the correct option (${key.toUpperCase()})`,
headingUrdu: q.portionHeadingUrdu || '',
items: []
};
}
map[key].items.push(q);
} else {
unassigned.push(q);
}
});

const groups = Object.keys(map).sort().map(k => map[k]);
if (unassigned.length > 0) {
groups.push({ portionKey: '', heading: 'Additional MCQs', headingUrdu: '', items: unassigned });
}
return groups;
};

const groupedData = getGroupedMcqs();
let content = '';
let globalIndex = 0;

groupedData.forEach(group => {
// Render Portion Header if portions exist
if (hasPortions && group.portionKey) {
const { eng: pEng, urdu: pUrdu } = getDisplayText(group.heading, group.headingUrdu, true);
if (isBilingual) {
content += `
<div class="portion-sub-heading">
<span contenteditable="true" data-portion-key="${group.portionKey}" data-type="portion-heading-eng"><strong class="portion-tag">(${group.portionKey})</strong> ${pEng}</span>
<span class="urdu-text" contenteditable="true" data-portion-key="${group.portionKey}" data-type="portion-heading-urdu" style="font-size:0.9em;"><strong class="portion-tag">(${group.portionKey})</strong> ${pUrdu}</span>
</div>`;
} else {
const pText = isUrduMode ? (pUrdu || pEng) : (pEng || pUrdu);
content += `
<div class="portion-sub-heading ${isUrduMode ? 'urdu-text' : ''}" style="text-align: ${isUrduMode ? 'right' : 'left'};">
<span contenteditable="true" data-portion-key="${group.portionKey}" data-type="portion-heading-${isUrduMode ? 'urdu' : 'eng'}">
<strong class="portion-tag">(${group.portionKey})</strong> ${pText}
</span>
</div>`;
}
}

const syllabusElem = document.querySelector('[data-field="syllabusText"]');
if (syllabusElem) paperSettings.syllabusText = syllabusElem.innerHTML.trim();

const testNoElem = document.querySelector('[data-field="testNoValue"]');

const urduSpan = container.querySelector('.urdu-text');

} else {
paperData[map.data] = container.innerHTML.trim();
}
}

const mcqSuffixElem = document.querySelector('[data-field="mcqSectionHeadingSuffix"]');

['shortQuestions', 'longQuestions'].forEach(key => {

paperData[key].forEach(q => {
// Save the Question Text
const engElem = document.querySelector(`[data-question-id="${q.id}"][data-type="${typePrefix}-question-eng"]`);
if (engElem) q.question = cleanQuestionText(engElem.innerHTML);

const urduElem = document.querySelector(`[data-question-id="${q.id}"][data-type="${typePrefix}-question-urdu"]`);
if (urduElem) q.questionUrdu = cleanQuestionText(urduElem.innerHTML);

// FIX: Save the Heading Text (The "Write short answers" part)
// Save the Heading Text — but ONLY as a manual override if it
// actually differs from what the app would auto-show right now
// (board-pattern text or the generic default). If it matches
// the auto text, clear any stored override instead of freezing
// it. This is what makes the useBoardHeadings checkbox apply
// instantly on Save & Apply: nothing is left pinned to stale
// text just because Settings happened to be opened once.
const headEng = document.querySelector(`[data-question-id="${q.id}"][data-type="${typePrefix}-heading-eng"]`);
const headUrdu = document.querySelector(`[data-question-id="${q.id}"][data-type="${typePrefix}-heading-urdu"]`);

if (paperSettings.useBoardHeadings) {
// Board Pattern Headings is the authority right now, so an
// edit here updates the board pattern itself — NOT the
// custom override (q.heading). This keeps the custom
// heading untouched underneath, ready to reappear once the
// checkbox is unticked.
if (headEng) {
const val = cleanQuestionText(headEng.innerHTML);
if (!paperData.boardPattern) paperData.boardPattern = {};
if (!paperData.boardPattern.questionHeadings) paperData.boardPattern.questionHeadings = {};
paperData.boardPattern.questionHeadings[q.questionNumber] = val;
}
if (headUrdu) {
const val = cleanQuestionText(headUrdu.innerHTML);
if (!paperData.boardPattern) paperData.boardPattern = {};
if (!paperData.boardPattern.questionHeadings) paperData.boardPattern.questionHeadings = {};
paperData.boardPattern.questionHeadings[q.questionNumber + "Urdu"] = val;
}
} else {
// Board Pattern Headings is OFF, so an edit here is a
// genuine custom override. Only store it as an override if
// it actually differs from the generic default — otherwise
// clear any old override so the default keeps showing.
const auto = computeAutoHeadingText(q.questionNumber, typePrefix);
if (headEng) {
const val = cleanQuestionText(headEng.innerHTML);
if (val && val !== auto.eng) {
q.heading = val;
} else {
delete q.heading;
}
}
if (headUrdu) {
const val = cleanQuestionText(headUrdu.innerHTML);
if (val && val !== auto.urdu) {
q.headingUrdu = val;
} else {
delete q.headingUrdu;
}
}
}
});
});

function processText(text = '') {
if (!text || typeof text !== 'string') return '';

// 1. Stash and protect all HTML tags (including existing <span>, <img>, etc.)
const tagStore = [];
let processed = text.replace(/<[^>]+>/g, (tag) => {
const placeholder = `__HTMLTAG_${tagStore.length}__`;
tagStore.push(tag);
return placeholder;
});

// 2. Handle Arabic markers: <!text<! or <!text!>
processed = processed.replace(/<!(.*?)<!/g, '<span class="arabic-text">$1</span>')
.replace(/<!(.*?)!>/g, '<span class="arabic-text">$1</span>');

// 3. Stash LaTeX blocks so regex doesn't alter formula contents
const mathStore = [];
const mathRegex = /(\$\$[\s\S]*?\$\$|\$.*?\$|\\\(.*?\\\)|\\\[.*?\\\])/g;
processed = processed.replace(mathRegex, (match) => {
const placeholder = `__MATH_${mathStore.length}__`;
mathStore.push(match);
return placeholder;
});

// 4. Wrap remaining standalone numbers/units in Urdu text with LTR

processed = processed.replace(ltrRegex, '<span class="ltr-in-rtl">$1</span>');

mathStore.forEach((math, idx) => {

});

// 6. Restore HTML tags

});

return processed;
}
function toRomanLower(num) {
if (num < 1) return num.toString();
const lookup = [
{ value: 50, symbol: 'l' },
{ value: 40, symbol: 'xl' },
{ value: 10, symbol: 'x' },
{ value: 9, symbol: 'ix' },
{ value: 5, symbol: 'v' },
{ value: 4, symbol: 'iv' },
{ value: 1, symbol: 'i' }
];

let result = '';
for (const item of lookup) {
while (num >= item.value) {
result += item.symbol;
num -= item.value;
}
}
return result;
}

// 3. Delegate drag-and-drop mouse/pointer events inside #paperContent
document.addEventListener('mousedown', function(e) {

if (!img) {
// Hide toolbar if clicking completely outside
if (!e.target.closest('#global-img-toolbar')) {
toolbar.style.display = 'none';
activeImg = null;
}
return;
}

e.preventDefault(); // Prevents cursor focus changes
activeImg = img;

// Store starting mouse coordinate vectors
const startX = e.clientX;
const startY = e.clientY;

// Read current CSS positional offset (or default to 0)
const startLeft = parseFloat(img.style.left) || 0;
const startTop = parseFloat(img.style.top) || 0;

let isDragging = false;
toolbar.style.display = 'none'; // Temporarily hide toolbar during drag movement

function onMouseMove(moveEvent) {
const dx = moveEvent.clientX - startX;
const dy = moveEvent.clientY - startY;

// Threshold of movement to distinguish a drag from a simple click
if (Math.abs(dx) > 3 || Math.abs(dy) > 3) {
isDragging = true;
}

if (isDragging) {
img.style.position = 'relative';
img.style.left = (startLeft + dx) + 'px';
img.style.top = (startTop + dy) + 'px';
}
}

function onMouseUp() {
document.removeEventListener('mousemove', onMouseMove);
document.removeEventListener('mouseup', onMouseUp);

if (isDragging) {
// Drag ended: show the toolbar at the new coordinates
positionToolbar(img);
if (typeof saveEditableContent === 'function') saveEditableContent(false);
} else {
// Simple click detected: toggle/position toolbar
positionToolbar(img);
}
}

document.addEventListener('mousemove', onMouseMove);
document.addEventListener('mouseup', onMouseUp);
});
